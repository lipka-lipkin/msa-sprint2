===== 2. Чужие бронирования: userid: user2 → FORBIDDEN
{
  "errors": [
    {
      "message": "Access denied: user user2 cannot view bookings of user user1",
      "path": [
        "bookingsByUser"
      ],
      "extensions": {
        "code": "FORBIDDEN",
        "serviceName": "booking"
      }
    }
  ],
  "data": {
    "bookingsByUser": null
  }
}

===== 3. Без заголовка userid → UNAUTHENTICATED
{
  "errors": [
    {
      "message": "Access denied: userid header is required",
      "path": [
        "bookingsByUser"
      ],
      "extensions": {
        "code": "UNAUTHENTICATED",
        "serviceName": "booking"
      }
    }
  ],
  "data": {
    "bookingsByUser": null
  }
}
